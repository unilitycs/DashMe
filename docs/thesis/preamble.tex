% 1) \todo{testo}: riquadro monocromatico per annotazioni o completamenti pendenti
\newsavebox{\todobox}
\newcommand{\todo}[1]{%
  \sbox{\todobox}{\small\textbf{[DA COMPLETARE:} #1\textbf{]}}%
  \ifdim\wd\todobox>0.88\linewidth
    \par\smallskip\noindent
    {%
      \setlength{\fboxsep}{5pt}%
      \fcolorbox{black!60}{black!5}{%
        \parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{%
          \small\color{black}\textbf{[DA COMPLETARE:} #1\textbf{]}%
        }%
      }%
    }%
    \par\smallskip
  \else
    \fcolorbox{black!60}{black!5}{%
      \small\color{black}\textbf{[DA COMPLETARE:} #1\textbf{]}%
    }%
  \fi
}

% 2) \dashboardfig[width]{filename}{caption}{label}:
%    cerca in images/, poi in figures/ e infine in ../Tableau/; se assente mostra un segnaposto pulito
\newcommand{\dashboardfig}[4][\textwidth]{%
  \begin{figure}[htbp]
    \centering
    \IfFileExists{images/#2}{%
      \includegraphics[width=#1,keepaspectratio]{images/#2}%
    }{%
      \IfFileExists{figures/#2}{%
        \includegraphics[width=#1,keepaspectratio]{figures/#2}%
      }{%
        \IfFileExists{../Tableau/#2}{%
          \includegraphics[width=#1,keepaspectratio]{../Tableau/#2}%
        }{%
          {%
            \setlength{\fboxsep}{10pt}%
            \setlength{\fboxrule}{0.8pt}%
            \fcolorbox{black!60}{black!4}{%
              \parbox{\dimexpr0.92\linewidth-2\fboxsep-2\fboxrule\relax}{%
                \centering\vspace{1.2em}
                {\large\bfseries [Figura segnaposto: \texttt{\detokenize{#2}}]}\\[0.6em]
                {\small Immagine non ancora esportata. Salvare il file della dashboard in \texttt{docs/thesis/images/\detokenize{#2}} (oppure in \texttt{docs/thesis/figures/\detokenize{#2}} o \texttt{docs/Tableau/\detokenize{#2}}), quindi ricompilare il documento.}%
                \vspace{1.2em}
              }%
            }%
          }%
        }%
      }%
    }%
    \caption{#3}%
    \label{#4}%
  \end{figure}%
}

% 3) \keyinsight{titolo}{corpo}: box di sintesi monocromatico
\newcommand{\keyinsight}[2]{%
  \par\medskip\noindent
  {%
    \setlength{\fboxsep}{8pt}%
    \setlength{\fboxrule}{0.6pt}%
    \fcolorbox{black}{black!3}{%
      \parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{%
        {\bfseries #1}\par\smallskip
        {\small #2}%
      }%
    }%
  }%
  \par\medskip
}
